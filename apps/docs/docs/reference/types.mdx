---
title: Types
description: The TypeScript types exported from the package root, which are shared by the server, tracker, client and dashboard.
---

The root export, `analytik`, contains types only. There is no runtime code in it, so importing from it adds nothing to a bundle.

```ts
import type { Overview, PaymentInput, PrivacyMode } from 'analytik'
```

## Unions

```ts
type PrivacyMode = 'strict' | 'enriched' | 'persistent'
type Interval = 'hour' | 'day' | 'week' | 'month'
type EventType = 'pageview' | 'goal' | 'outbound' | 'ping' | 'identify' | 'revenue'
type PaymentKind = 'payment' | 'refund'
type PaymentMatch = 'metadata' | 'return_page' | 'customer' | 'email' | 'timing'
type PaymentProvider = 'stripe' | 'dodo' | 'lemonsqueezy'

type Dimension =
  | 'channel'
  | 'source'
  | 'referrer'
  | 'campaign'
  | 'medium'
  | 'term'
  | 'content'
  | 'ref'
  | 'page'
  | 'entry'
  | 'exit'
  | 'country'
  | 'region'
  | 'city'
  | 'browser'
  | 'os'
  | 'device'
  | 'event'
  | 'outbound'
  | 'goal'

type Props = Record<string, string | number | boolean | null>
```

## Site

```ts
interface Site {
  id: string
  name: string
  domain: string
  timezone: string
  currency: string
  privacyMode: PrivacyMode
  retentionDays: number | null
  allowLocalhost: boolean
  ignoredIps: string[]
  createdAt: number // Unix milliseconds
  /** Which providers have a webhook secret configured. Secrets are never returned. */
  webhooks?: Record<PaymentProvider, boolean>
  /** Token for the read-only share link, or null when sharing is off. */
  shareToken?: string | null
}
```

## Events and payments

```ts
/** What the tracker posts to `POST {basePath}/event`. */
interface TrackerPayload {
```

## Queries and results

`Overview.revenue`, `orders` and the other revenue figures count only payments in the site's own `currency`. Payments in any other currency are listed in `otherCurrencies`, not converted.

```ts
interface Filters {
  [dimension: string]: string | undefined
}

interface Query {
  from: number // Unix ms, inclusive
  to: number // Unix ms, exclusive
  interval?: Interval
  tz?: string
  filters?: Filters
}

interface Delta {
  value: number
  previous: number
  /** Relative change, `null` when `previous` is 0. */
  change: number | null
}

interface Overview {
  visitors: Delta
  sessions: Delta
  pageviews: Delta
  bounceRate: Delta
  avgDurationMs: Delta
  revenue: Delta
  orders: Delta
  conversionRate: Delta
  revenuePerVisitor: Delta
  currency: string
  online: number
  /** Payments in other currencies than the site's, left out of the revenue numbers above. */
  otherCurrencies: { currency: string; amount: number; orders: number }[]
}

interface TimeseriesPoint {
  ts: number // bucket start, Unix ms
  visitors: number
  sessions: number
  pageviews: number
  revenue: number
}

interface BreakdownRow {
  key: string
  visitors: number
  sessions: number
  pageviews: number
  revenue: number
  orders: number
  conversionRate: number
}

interface Realtime {
  online: number
  recent: { ts: number; type: EventType; name: string | null; path: string | null; country: string | null }[]
  locations: { lat: number; lon: number; country: string | null; city: string | null }[]
}

interface SessionSummary {
  id: string
  visitorId: string
  startedAt: number
  durationMs: number
  pageviews: number
  goals: number
  entryPage: string | null
  exitPage: string | null
  source: string
  channel: string
  country: string | null
  city: string | null
  browser: string | null
  os: string | null
  device: string | null
  revenue: number
}

interface Page<T> {
  items: T[]
  nextCursor: string | null
}
```

## Funnels

```ts
interface Funnel {
  id: string
  name: string
  steps: FunnelStep[]
}

interface FunnelStep {
  type: 'page' | 'goal'
  /** Page path (supports a trailing `*`) or goal name. */
  match: string
}

interface FunnelResult {
  funnel: Funnel
  steps: { match: string; visitors: number; dropOff: number | null }[]
}
```
